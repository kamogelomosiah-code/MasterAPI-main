Quick start

# 1. Create an app (admin, once)
curl -X POST $API/api/admin/apps -H "Authorization: Bearer $MASTER_KEY" \
  -H "Content-Type: application/json" -d '{"name":"My shop"}'
# → { "apiKey": "mk_live_…" }   save it, it's shown once

# 2. Use it from your app
curl -X POST $API/api/v1/data/products -H "X-API-Key: mk_live_…" \
  -H "Content-Type: application/json" -d '{"name":"Pen","price":10}'
curl "$API/api/v1/data/products?price=10&sort=-createdAt&limit=20" -H "X-API-Key: mk_live_…"

From JavaScript

const api = (path, opts = {}) => fetch(`${API}/api/v1${path}`, {
  ...opts,
  headers: { "X-API-Key": KEY, "Content-Type": "application/json", ...opts.headers },
}).then((r) => r.json());

await api("/data/todos", { method: "POST", body: JSON.stringify({ text: "Ship it" }) });
const { data } = await api("/data/todos?done=false");
